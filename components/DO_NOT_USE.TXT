# Optional dependency
# - ccache

# Safety net, exit if prompts fail
set -e

say "Step [1] of [5]-> Preparing build environment... "
# Remove any existing build dir for fresh compile
deletedir-dev -f build
newdir-dev build
goto build

say "Step [2] of [5]-> Configuring build with CMake..."
# Using ccache by default
cmake -DCMAKE_C_COMPILER_LAUNCHER=ccache ..
# If ccache not used, comment out above prompt and uncomment below prompt
#cmake ..

say "Step [3] of [5]-> Compiling source code..."
make

say "Step [4] of [5]-> Installing 'screen' (elevate required)..."
# Prompt for elevated privs only at the install step
elevate make install

say "Step [5] of [5]-> Cleaning up build environment..."
goto ..
deletedir-dev build

# Confirmation
say "[SUCCESS] The build environment is clean, and 'screen' is installed and ready to use!"
